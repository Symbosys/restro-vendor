// ==========================================
// 📦 PACKING MATERIAL ORDERING SYSTEM
// ==========================================
// Vendors can order packing materials with 3-week installment payment via wallet

// ==========================================
// 1. PACKING PRODUCT CATALOG (No Categories)
// ==========================================

model PackingProduct {
  id              String   @id @default(uuid())
  title           String   // "Kraft Paper Box", "Brown Paper Bag"
  description     String   // "Eco-friendly paper container 500ml"
  
  // Pricing
  price           Decimal  @db.Decimal(10, 2) // Price per unit pack
  quantityPerPack Int      // 50 pcs, 100 pcs, etc.
  minOrderQty     Int      @default(1) // Minimum order quantity
  maxOrderQty     Int?     // Maximum order quantity (optional)
  
  // Product Details
  imageUrl        String?
  badge           String?  // "Eco-friendly", "Premium Quality"
  badgeColor      String?  // Hex color code
  sku             String?  @unique // Stock Keeping Unit
  
  // Inventory Management
  stockQuantity   Int      @default(0)
  lowStockAlert   Int      @default(10)
  isAvailable     Boolean  @default(true)
  isActive        Boolean  @default(true)
  
  // Metadata
  weight          Decimal? @db.Decimal(8, 2) // in grams
  dimensions      String?  // "10x10x5 cm"
  material        String?  // "Kraft Paper", "Plastic", etc.
  displayOrder    Int      @default(0) // For sorting in app
  
  orderItems      PackingOrderItem[]
  cartItems       PackingCartItem[]
  
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  
  @@index([isActive, isAvailable])
  @@index([displayOrder])
  @@map("packing_products")
}

// ==========================================
// 2. SHOPPING CART
// ==========================================

model PackingCart {
  id          String   @id @default(uuid())
  vendorId    BigInt   @unique // One cart per vendor
  
  items       PackingCartItem[]
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  vendor      Vendor   @relation(fields: [vendorId], references: [id], onDelete: Cascade)
  
  @@index([vendorId])
  @@map("packing_carts")
}

model PackingCartItem {
  id          String   @id @default(uuid())
  
  cartId      String
  cart        PackingCart @relation(fields: [cartId], references: [id], onDelete: Cascade)
  
  productId   String
  product     PackingProduct @relation(fields: [productId], references: [id])
  
  quantity    Int      // Number of packs
  priceAtAdd  Decimal  @db.Decimal(10, 2) // Price when added to cart
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@unique([cartId, productId])
  @@index([cartId])
  @@map("packing_cart_items")
}

// ==========================================
// 3. PACKING ORDERS
// ==========================================

enum PackingOrderStatus {
  PENDING           // Order placed, awaiting processing
  CONFIRMED         // Order confirmed
  PROCESSING        // Being prepared
  SHIPPED           // Shipped to vendor
  DELIVERED         // Delivered successfully
  CANCELLED         // Order cancelled
  FAILED            // Payment or other failure
}

model PackingOrder {
  id              String   @id @default(uuid())
  orderNumber     String   @unique // "PKG-2025-014"
  
  vendorId        BigInt
  vendor          Vendor   @relation(fields: [vendorId], references: [id], onDelete: Cascade)
  
  // Order Details
  status          PackingOrderStatus @default(PENDING)
  totalAmount     Decimal  @db.Decimal(10, 2) // Total order amount
  itemsCount      Int      // Total number of items
  
  // Delivery Information
  deliveryAddress String?
  deliveryNotes   String?
  estimatedDelivery DateTime?
  actualDelivery  DateTime?
  
  // Payment Information
  paymentMethod   String   @default("WALLET_INSTALLMENT") // Always wallet installment
  isPaid          Boolean  @default(false)
  paidAmount      Decimal  @db.Decimal(10, 2) @default(0)
  pendingAmount   Decimal  @db.Decimal(10, 2)
  
  // Relations
  items           PackingOrderItem[]
  installmentPlan PackingInstallmentPlan?
  
  // Timestamps
  confirmedAt     DateTime?
  cancelledAt     DateTime?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  
  @@index([vendorId, status])
  @@index([vendorId, createdAt])
  @@index([orderNumber])
  @@map("packing_orders")
}

model PackingOrderItem {
  id          String   @id @default(uuid())
  
  orderId     String
  order       PackingOrder @relation(fields: [orderId], references: [id], onDelete: Cascade)
  
  productId   String
  product     PackingProduct @relation(fields: [productId], references: [id])
  
  quantity    Int      // Number of packs ordered
  pricePerUnit Decimal @db.Decimal(10, 2) // Price at time of order
  totalPrice  Decimal  @db.Decimal(10, 2) // quantity * pricePerUnit
  
  // Product snapshot (in case product details change later)
  productSnapshot Json? // Store product details at time of order
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@index([orderId])
  @@index([productId])
  @@map("packing_order_items")
}

// ==========================================
// 4. INSTALLMENT PAYMENT SYSTEM (3-Week Plan)
// ==========================================

enum InstallmentStatus {
  PENDING       // Not yet due
  UPCOMING      // Due soon (within 2-3 days)
  DUE           // Due now
  PROCESSING    // Payment being processed
  PAID          // Successfully paid
  OVERDUE       // Payment missed
  FAILED        // Payment failed
  CANCELLED     // Installment cancelled
}

model PackingInstallmentPlan {
  id              String   @id @default(uuid())
  
  orderId         String   @unique
  order           PackingOrder @relation(fields: [orderId], references: [id], onDelete: Cascade)
  
  vendorId        BigInt
  vendor          Vendor   @relation(fields: [vendorId], references: [id], onDelete: Cascade)
  
  // Plan Details
  totalAmount     Decimal  @db.Decimal(10, 2) // Total amount to be paid
  installmentCount Int     @default(3) // Number of installments (always 3)
  installmentAmount Decimal @db.Decimal(10, 2) // Amount per installment
  
  // Tracking
  paidInstallments Int     @default(0) // Number of installments paid
  remainingAmount  Decimal @db.Decimal(10, 2) // Amount still pending
  
  // Status
  isCompleted     Boolean  @default(false)
  isCancelled     Boolean  @default(false)
  
  // Relations
  installments    PackingInstallment[]
  
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  
  @@index([vendorId, isCompleted])
  @@index([orderId])
  @@map("packing_installment_plans")
}

model PackingInstallment {
  id              String   @id @default(uuid())
  
  planId          String
  plan            PackingInstallmentPlan @relation(fields: [planId], references: [id], onDelete: Cascade)
  
  // Installment Details
  installmentNumber Int    // 1, 2, or 3
  amount          Decimal  @db.Decimal(10, 2) // Amount for this installment
  status          InstallmentStatus @default(PENDING)
  
  // Due Date Management
  dueDate         DateTime // When this installment is due
  scheduledDate   DateTime // When auto-deduction is scheduled
  paidDate        DateTime? // Actual payment date
  
  // Payment Information
  paymentMethod   String?  @default("WALLET") // Wallet deduction
  transactionId   String?  @unique // Reference to VendorWalletTransaction
  
  // Failed Payment Tracking
  attemptCount    Int      @default(0)
  lastAttemptAt   DateTime?
  failureReason   String?
  
  // Notifications
  reminderSentAt  DateTime? // When reminder notification was sent
  
  // Relations
  transaction     VendorWalletTransaction? @relation(fields: [transactionId], references: [id])
  
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  
  @@unique([planId, installmentNumber])
  @@index([planId, status])
  @@index([dueDate, status])
  @@index([scheduledDate, status])
  @@map("packing_installments")
}

/*
1. NO CATEGORIES - Products directly listed (as per user request)
2. Uses existing VendorWallet for payments
3. Automatic weekly deduction via cron job
4. Example: ₹600 order = 3 installments of ₹200 each
5. Payment flow:
   - Week 1: Create PackingInstallment (installmentNumber: 1, amount: ₹200, dueDate: Week 1)
   - Week 2: Create PackingInstallment (installmentNumber: 2, amount: ₹200, dueDate: Week 2)
   - Week 3: Create PackingInstallment (installmentNumber: 3, amount: ₹200, dueDate: Week 3)
   
6. Cron job logic:
   - Find installments where dueDate <= today AND status IN ['DUE', 'OVERDUE']
   - Check VendorWallet balance >= installment.amount
   - If yes:
     * Create VendorWalletTransaction (DEBIT)
     * Update installment status to PAID
     * Update plan paidInstallments counter
     * Send PACKING_INSTALLMENT_PAID notification
   - If no:
     * Update installment status to FAILED
     * Increment attemptCount
     * Send PACKING_PAYMENT_FAILED notification
     
7. Key indexes for performance:
   - vendorId + status (fast vendor queries)
   - dueDate + status (cron job processing)
   - scheduledDate + status (scheduled payments)
*/
